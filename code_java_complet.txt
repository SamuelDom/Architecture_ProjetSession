    private static final class JsonObject {
        private final String source;
        private int index;

        private JsonObject(String source) {
            this.source = source;
        }

        static Map<String, Object> parse(String source) {
            return new JsonObject(source).parseObject();
        }

        private Map<String, Object> parseObject() {
            Map<String, Object> values = new HashMap<>();
            skipWhitespace();
            expect('{');
            skipWhitespace();
            if (consume('}')) {
                ensureFinished();
                return values;
            }
            while (true) {
                skipWhitespace();
                String key = parseString();
                skipWhitespace();
                expect(':');
                skipWhitespace();
                Object value = peek() == '"' ? parseString() : parseInteger();
                if (values.putIfAbsent(key, value) != null) {
                    throw invalidJson();
                }
                skipWhitespace();
                if (consume('}')) {
                    ensureFinished();
                    return values;
                }
                expect(',');
            }
        }

        private String parseString() {
            expect('"');
            StringBuilder value = new StringBuilder();
            while (index < source.length()) {
                char character = source.charAt(index++);
                if (character == '"') {
                    return value.toString();
                }
                if (character < 0x20) {
                    throw invalidJson();
                }
                if (character != '\\') {
                    value.append(character);
                    continue;
                }
                if (index >= source.length()) {
                    throw invalidJson();
                }
                char escaped = source.charAt(index++);
                switch (escaped) {
                    case '"', '\\', '/' -> value.append(escaped);
                    case 'b' -> value.append('\b');
                    case 'f' -> value.append('\f');
                    case 'n' -> value.append('\n');
                    case 'r' -> value.append('\r');
                    case 't' -> value.append('\t');
                    case 'u' -> value.append(parseUnicodeEscape());
                    default -> throw invalidJson();
                }
            }
            throw invalidJson();
        }

        private char parseUnicodeEscape() {
            if (index + 4 > source.length()) {
                throw invalidJson();
            }
            try {
                char value = (char) Integer.parseInt(source.substring(index, index + 4), 16);
                index += 4;
                return value;
            } catch (NumberFormatException exception) {
                throw invalidJson();
            }
        }

        private Integer parseInteger() {
            int start = index;
            if (peek() == '-') {
                index++;
            }
            while (index < source.length() && Character.isDigit(source.charAt(index))) {
                index++;
            }
            if (start == index || (source.charAt(start) == '-' && start + 1 == index)) {
                throw invalidJson();
            }
            try {
                return Integer.valueOf(source.substring(start, index));
            } catch (NumberFormatException exception) {
                throw invalidJson();
            }
        }

        private void skipWhitespace() {
            while (index < source.length()) {
                char character = source.charAt(index);
                if (character != ' ' && character != '\n' && character != '\r' && character != '\t') {
                    return;
                }
                index++;
            }
        }

        private char peek() {
            return index < source.length() ? source.charAt(index) : '\0';
        }

        private boolean consume(char expected) {
            if (peek() != expected) {
                return false;
            }
            index++;
            return true;
        }

        private void expect(char expected) {
            if (!consume(expected)) {
                throw invalidJson();
            }
        }

        private void ensureFinished() {
            skipWhitespace();
            if (index != source.length()) {
                throw invalidJson();
            }
        }

        private IllegalArgumentException invalidJson() {
            return new IllegalArgumentException("Le corps doit Ãªtre un objet JSON valide.");
        }
    }

public final class LobbyHttpHandler implements HttpHandler {

    private final CreateLobbyUseCase createLobbyUseCase;

    public LobbyHttpHandler(CreateLobbyUseCase createLobbyUseCase) {
        this.createLobbyUseCase = createLobbyUseCase;
    }

    @Override
    public void handle(HttpExchange exchange) throws IOException {
        String path = exchange.getRequestURI().getPath();
        String method = exchange.getRequestMethod();

        try {
            if ("POST".equals(method) && "/lobbies".equals(path)) {
                createLobby(exchange);
                return;
            }
            if ("GET".equals(method) || "POST".equals(method)) {
                sendError(exchange, 404, "NOT_FOUND", "Route introuvable.");
                return;
            }
            exchange.getResponseHeaders().set("Allow", "POST");
            sendError(exchange, 405, "METHOD_NOT_ALLOWED", "MÃ©thode non autorisÃ©e.");
        } catch (AlreadyInLobbyException exception) {
            sendError(exchange, 409, "ALREADY_IN_LOBBY", exception.getMessage());
        } catch (DomainException | IllegalArgumentException exception) {
            sendError(exchange, 400, "BAD_REQUEST", exception.getMessage());
        }
    }

    private void createLobby(HttpExchange exchange) throws IOException {
        String playerId = exchange.getRequestHeaders().getFirst("X-Player-Id");
        LobbyOutput lobby = createLobbyUseCase.execute(new CreateLobbyCommand(playerId));
        exchange.getResponseHeaders().set("Location", "/lobbies/" + lobby.id());
        sendJson(exchange, 201, serialize(lobby));
    }

    private static String serialize(LobbyOutput lobby) {
        String membersJson = lobby.members().stream()
                .map(LobbyHttpHandler::quote)
                .collect(Collectors.joining(",", "[", "]"));

        return "{"
                + "\"id\":" + quote(lobby.id()) + ","
                + "\"hostId\":" + quote(lobby.hostId()) + ","
                + "\"members\":" + membersJson + ","
                + "\"availableSeats\":" + lobby.availableSeats() + ","
                + "\"status\":" + quote(lobby.status())
                + "}";
    }

    private static String quote(String value) {
        StringBuilder json = new StringBuilder("\"");
        for (int index = 0; index < value.length(); index++) {
            char character = value.charAt(index);
            switch (character) {
                case '"' -> json.append("\\\"");
                case '\\' -> json.append("\\\\");
                case '\b' -> json.append("\\b");
                case '\f' -> json.append("\\f");
                case '\n' -> json.append("\\n");
                case '\r' -> json.append("\\r");
                case '\t' -> json.append("\\t");
                default -> {
                    if (character < 0x20) {
                        json.append(String.format("\\u%04x", (int) character));
                    } else {
                        json.append(character);
                    }
                }
            }
        }
        return json.append('"').toString();
    }

    private static void sendError(HttpExchange exchange, int status, String code, String message) throws IOException {
        String body = "{"
                + "\"error\":" + quote(code) + ","
                + "\"message\":" + quote(message == null ? "RequÃªte invalide." : message)
                + "}";
        sendJson(exchange, status, body);
    }

    private static void sendJson(HttpExchange exchange, int status, String body) throws IOException {
        byte[] response = body.getBytes(StandardCharsets.UTF_8);
        exchange.getResponseHeaders().set("Content-Type", "application/json; charset=utf-8");
        exchange.sendResponseHeaders(status, response.length);
        try (OutputStream output = exchange.getResponseBody()) {
            output.write(response);
        }
    }

        if (participants.stream().anyMatch(Objects::isNull)) {
            throw new IllegalArgumentException("Un participant ne peut pas Ãªtre null.");

public final class LevelTable {
    public static final int MAX_LEVEL = 5;
    private static final int[] XP_REQUIRED = {0, 300, 900, 2700, 6500};

    public int levelForXp(int totalXp) {
        if (totalXp < 0) throw new IllegalArgumentException("L'expÃ©rience ne peut pas Ãªtre nÃ©gative.");
        int level = 1;
        for (int index = 0; index < XP_REQUIRED.length; index++) {
            if (totalXp >= XP_REQUIRED[index]) level = index + 1;
        }
        return level;
    }

    public int xpRequiredForLevel(int level) {
        if (level < 1 || level > MAX_LEVEL) {
            throw new IllegalArgumentException("Le niveau doit Ãªtre entre 1 et " + MAX_LEVEL + ".");
        }
        return XP_REQUIRED[level - 1];
    }
}

    }

    public LevelUpResult checkLevelUp(Hero hero) {
        Objects.requireNonNull(hero, "Le hÃ©ros est requis.");
        int oldLevel = hero.level();
        int newLevel = levelTable.levelForXp(hero.getTotalXp());
        int hpGained = oldLevel < newLevel ? hero.applyLevelUp(newLevel) : 0;
